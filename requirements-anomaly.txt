#   pip install -r requirements-anomaly.txt
#   python detect_anomalies.py --raw ./data/raw --out ./artifacts

#   artifacts/      <- created by the script; the 3 output CSVs land here.

# render_offline_graph.py) are a separate lineage-graph track and additionally
